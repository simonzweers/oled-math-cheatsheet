\section{Calculus}
\begin{definition}[Derivative]
	\\
	$\displaystyle f'(x) = \lim_{h \to 0} \dfrac{f(x+h) - f(x)}{h} = \frac{d}{dx}f(x)$ \\
	Differentiation being a linear operation.
\end{definition}

\begin{example}
	List of derivatives and antiderivatives:

	\vspace{2mm}
	\begin{tabular}{lcc}
		\toprule
		$f(x)$ & $f'(x)$ & $F(x)$ \\

		$\sin(x)$	& $\cos(x)$		& $-\cos(x)$ \\

\end{example}

\begin{example}
	Differentiation/Integration rules:

	Chain rule: $(f \circ g)'(x) = [f(g(x))]' = f'(g(x))g'(x)$ 

	Product rule: $[f(x)g(x)]' = f'(x)g(x) + f(x)g'(x) $ 

	Quotient rule: $\dfrac{f(x)}{g(x)} = \dfrac{f'(x)g(x) - f(x)g'(x)}{(g(x))^2} $ 
	
\end{example}
